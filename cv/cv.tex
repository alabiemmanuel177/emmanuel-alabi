\cvsection{Research}

\cventry{Risk-Calibrated Semantic Navigation Under Distribution Shift}{2026}
{\small Preregistered ROS 2 / Gazebo / Nav2 benchmark.
\href{https://doi.org/10.5281/zenodo.22181941}{doi.org/10.5281/zenodo.22181941}}
\begin{cvitems}
  \item Calibration improved and clean-route efficiency held, but the
        preregistered collision-reduction and severity hypotheses did not:
        better component calibration did not establish safer navigation.
  \item Four systems differing only in the treatment component; protocol,
        manifest, thresholds and analysis code hash-locked before execution.
        6,840 episodes; independent clean-clone reproduction byte-identical.
  \item \href{https://www.emmanuelalabi.com/research/risk-calibrated-semantic-navigation}{emmanuelalabi.com/research/risk-calibrated-semantic-navigation}
\end{cvitems}

\cventry{Driver Drowsiness Detection --- Reproduction \& Evaluation Study}{2026}
{\small Independent computer vision study.
\href{https://github.com/alabiemmanuel177/driver-drowsiness-reproduction}{github.com/alabiemmanuel177/driver-drowsiness-reproduction}}
\begin{cvitems}
  \item An oracle-classifier decomposition showed the historical temporal
        threshold, not residual classification error, dominated false alerts
        once classifier accuracy was sufficient.
  \item Subject-disjoint reproduction on the MRL Eye Dataset; CNN trained from
        scratch against a HOG+SVM baseline, with deterministic splits, automated
        leakage checks and three-seed ablations.
  \item \href{https://www.emmanuelalabi.com/projects/driver-drowsiness-reproduction}{emmanuelalabi.com/projects/driver-drowsiness-reproduction}
\end{cvitems}
